▲ Chapter 01 — The Problem

Underground.
60 tonnes.
Zero visibility.

In underground coal and ore mines, dust clouds, methane gas, water mist and dense fog reduce visibility to near-zero. A driver of a 60-tonne ore hauler cannot see more than 1–2 metres ahead. At those conditions, standard distance sensors fail.

Even if an obstacle is detected 2 metres away, the vehicle needs several more metres of braking distance if moving at speed. OREGUARD solves this by dynamically fusing optical visibility, proximity, and velocity into a single real-time risk score every 500ms.

// ESP32 Joystick Teleoperation Controller
#include <WiFi.h>

const char* ssid = "HelloESP32";
const char* password = "12345678";
const char* vehicleIP = "192.168.4.1";
const int vehiclePort = 3333;

#define VRX 34   // Joystick X-Axis (Left/Right)
#define VRY 35   // Joystick Y-Axis (Forward/Back)
#define SW  32   // Tactile Pushbutton (E-Stop)

const int DEADZONE = 300;
unsigned long lastSend = 0;

void loop() {
  if (!client.connected()) { connectVehicle(); return; }
  
  int x = analogRead(VRX);
  int y = analogRead(VRY);
  int button = digitalRead(SW);
  char command = 'S';

  if (button == LOW)               command = 'S';
  else if (y > centerY + DEADZONE) command = 'F'; // Forward
  else if (y < centerY - DEADZONE) command = 'B'; // Reverse
  else if (x < centerX - DEADZONE) command = 'L'; // Turn Left
  else if (x > centerX + DEADZONE) command = 'R'; // Turn Right
  else                             command = 'S'; // Neutral Stop

  if (millis() - lastSend >= 50) {
    client.write(command);
    lastSend = millis();
  }
}
